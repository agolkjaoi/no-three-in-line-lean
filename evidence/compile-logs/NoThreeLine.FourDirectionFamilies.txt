NoThreeLine.FiniteDirections.mem_reducedFamilyLabels (n : ℕ) (z : FamilyLabel) :
  z ∈ reducedFamilyLabels n ↔
    1 ≤ z.2.2 ∧ z.2.2 < n ∧ 1 ≤ z.2.1 ∧ z.2.1 ≤ z.2.2 ∧ z.2.1.Coprime z.2.2 ∧ (↑z.1 < 2 ∨ z.2.1 < z.2.2)

  {d ∈ directionIndices n | d.snd.1 ≠ 0 ∧ d.snd.2 ≠ 0} ⊆ Finset.image familyDirection (reducedFamilyLabels n)
